% Shared thesis preamble for the ADNU TinyML Smart Plug manuscript
% Compile with XeLaTeX in MiKTeX/TeX Live for Times New Roman support.
\documentclass[12pt,oneside]{report}

% ADNU page setup: top 1 in, bottom 1 in, right 1 in, left 1.25 in.
\usepackage[top=1in,bottom=1in,left=1.25in,right=1in]{geometry}

% Font setup: XeLaTeX/LuaLaTeX uses Times New Roman; pdfLaTeX falls back to Times-like fonts.
\usepackage{iftex}
\ifPDFTeX
  \usepackage[T1]{fontenc}
  \usepackage[utf8]{inputenc}
  \usepackage{newtxtext,newtxmath}
\else
  \usepackage{fontspec}
  \setmainfont{Times New Roman}
\fi

% Spacing and paragraph format.
\usepackage{setspace}
\doublespacing
\setlength{\parindent}{1.5cm}
\setlength{\parskip}{0pt}
\usepackage{indentfirst}
\emergencystretch=3em

% Heading formats requested by the adviser/user:
% 1st level: 13 pt bold; 2nd level: 12 pt bold; 3rd level: 12 pt regular; 4th level: 12 pt italic.
\usepackage{titlesec}
\titleformat{\chapter}[display]
  {\normalfont\bfseries\centering\fontsize{13}{15.6}\selectfont}
  {\chaptertitlename\ \thechapter}
  {0.25\baselineskip}
  {\MakeUppercase}
\titleformat{name=\chapter,numberless}[display]
  {\normalfont\bfseries\centering\fontsize{13}{15.6}\selectfont}
  {}
  {0pt}
  {\MakeUppercase}
\titleformat{\section}
  {\normalfont\bfseries\fontsize{13}{15.6}\selectfont}
  {\thesection}{1em}{}
\titleformat{\subsection}
  {\normalfont\bfseries\fontsize{12}{14.4}\selectfont}
  {\thesubsection}{1em}{}
\titleformat{\subsubsection}
  {\normalfont\fontsize{12}{14.4}\selectfont}
  {\thesubsubsection}{1em}{}
\titleformat{\paragraph}[runin]
  {\normalfont\itshape\fontsize{12}{14.4}\selectfont}
  {\theparagraph}{1em}{}
\titlespacing*{\chapter}{0pt}{0pt}{0.75\baselineskip}
\titlespacing*{\section}{0pt}{18pt}{0.25\baselineskip}
\titlespacing*{\subsection}{0pt}{12pt}{0.25\baselineskip}
\titlespacing*{\subsubsection}{0pt}{12pt}{0.25\baselineskip}
\titlespacing*{\paragraph}{0pt}{12pt}{1em}
\setcounter{secnumdepth}{4}
\setcounter{tocdepth}{3}
